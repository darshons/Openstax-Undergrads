from manim_voiceover import VoiceoverScene

        title = title_card("Glucose Metabolism", "how insulin moves sugar into cells")

        axes.move_to(grid_to_point("C2-E5"))
        x_lbl = label("time (min)", size=22).next_to(axes, DOWN, buff=0.2)
        curve = axes.plot(lambda x: 8 * np.exp(-0.5 * x) + 1, color=PALETTE["primary"])
        self.play(Create(axes), Write(x_lbl), run_time=1.2)

        # --- Narration beat 1 (narrator) ---
        service.set_voice(VOICE_MAP["narrator"])
        with self.voiceover(text="After a meal, blood glucose rises, then insulin brings it back down.") as tracker:

            peak = Dot(axes.c2p(1, 6), color=PALETTE["accent"])

            self.wait(max(0.0, tracker.duration - 2.0))

        # --- Narration beat 2 (highlight the key idea) ---
        service.set_voice(VOICE_MAP["narrator"])

            self.play(Create(box), run_time=0.6)
            self.wait(max(0.0, tracker.duration - 0.6))

        # --- End: clear the frame (Group, NOT VGroup) ---
        self.play(FadeOut(Group(*self.mobjects)))
